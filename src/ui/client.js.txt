// ------------------------------------------------------------------ formatting
const fmtUsd = (v, d) => {
  if (v == null || !isFinite(v)) return '—';
  const a = Math.abs(v);
  const dec = d ?? (a >= 1000 ? 2 : a >= 1 ? 2 : a >= 0.01 ? 4 : 8);
  return (v < 0 ? '-$' : '$') + a.toLocaleString('en-US', { minimumFractionDigits: dec, maximumFractionDigits: dec });
};
const fmtPx = (v) => {
  if (v == null) return '—';
  const a = Math.abs(v);
  const d = a >= 1000 ? 2 : a >= 1 ? 2 : a >= 0.01 ? 4 : a >= 0.0001 ? 6 : 9;
  return '$' + v.toLocaleString('en-US', { minimumFractionDigits: d, maximumFractionDigits: d });
};
const fmtPct = (v, d = 2) => (v == null || !isFinite(v) ? '—' : `${v >= 0 ? '+' : ''}${v.toFixed(d)}%`);
const cls = (v) => (v > 0 ? 'up' : v < 0 ? 'down' : '');
const fmtN = (v, d = 4) => (v == null ? '—' : Number(v).toLocaleString('en-US', { maximumFractionDigits: d }));
const fmtTime = (t) => new Date(t).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' });
const fmtDT = (t) => new Date(t).toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
const fmtAge = (h) => (h < 48 ? `${Math.floor(h)}h` : `${Math.floor(h / 24)}d`);
const short = (a) => (a ? `${a.slice(0, 6)}…${a.slice(-4)}` : '');
const mkLabel = (m) => { const [k, s] = String(m).split(':'); return k === 'perp' ? `${s}-PERP` : s; };

async function getJSON(url, opt) {
  const r = await fetch(url, { credentials: 'same-origin', ...opt });
  const j = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(j.error || `Request failed (${r.status})`);
  return j;
}
const postJSON = (url, body, method = 'POST') => getJSON(url, { method, headers: { 'content-type': 'application/json' }, body: JSON.stringify(body || {}) });

async function connect() {
  try {
    provider = await pickProvider();
    const [address] = await provider.request({ method: 'eth_requestAccounts' });
    const { getAddress } = await V();
    const addr = getAddress(address);
    const { message } = await postJSON('/api/auth/nonce', { address: addr });
    const signature = await provider.request({ method: 'personal_sign', params: [toHexUtf8(message), addr] });
    session = await postJSON('/api/auth/verify', { address: addr, message, signature });
    renderLogin();
    provider.on?.('accountsChanged', (a) => { if (session && (!a[0] || a[0].toLowerCase() !== session.address.toLowerCase())) { toast('<b>Wallet account changed — please sign in again.</b>'); session = null; renderLogin(); } });
    document.dispatchEvent(new CustomEvent('orla:session'));
    return session;
  } catch (e) {
    if (e.message !== 'cancelled' && e.message !== 'no wallet') toast(`<b>Could not sign in</b><div class="m">${esc(e.shortMessage || e.message)}</div>`);
    throw e;
  }
}

// wallet client on Robinhood Chain
async function wallet() {
  await ensureSession();
  if (!provider) provider = await pickProvider();
  const v = await V();
  const a = await abi();
  const chain = v.defineChain({ id: 4663, name: 'Robinhood Chain', nativeCurrency: { name: 'Ether', symbol: 'ETH', decimals: 18 }, rpcUrls: { default: { http: [a.net.rpc] } }, blockExplorers: { default: { name: 'Blockscout', url: a.net.explorer } }, contracts: { multicall3: { address: '0xcA11bde05977b3631167028862bE2a173976CA11' } } });
  const hex = '0x' + (4663).toString(16);
  const cur = await provider.request({ method: 'eth_chainId' });
  if (cur !== hex) {
    try { await provider.request({ method: 'wallet_switchEthereumChain', params: [{ chainId: hex }] }); }
    catch (e) {
      if (e.code === 4902 || /Unrecognized|not added|unknown/i.test(e.message)) await provider.request({ method: 'wallet_addEthereumChain', params: [{ chainId: hex, chainName: 'Robinhood Chain', nativeCurrency: { name: 'Ether', symbol: 'ETH', decimals: 18 }, rpcUrls: [a.net.rpc], blockExplorerUrls: [a.net.explorer] }] });
      else throw e;
    }
  }
  const [account] = await provider.request({ method: 'eth_requestAccounts' });
  if (account.toLowerCase() !== session.address.toLowerCase()) throw new Error('Wallet account changed — please sign in again.');
  const w = v.createWalletClient({ account: v.getAddress(account), chain, transport: v.custom(provider) });
  const p = v.createPublicClient({ chain, transport: v.http(a.net.rpc) });
  return { w, p, v, a, chain, account: v.getAddress(account) };
}

// run a transaction with feedback
async function tx(label, fn) {
  toast(`<b><span class="spin"></span> ${esc(label)}</b><div class="m">Confirm in your wallet…</div>`, 0);
  try {
    const { w, p } = await wallet();
    const hash = await fn(w, p);
    toast(`<b><span class="spin"></span> ${esc(label)}</b><div class="m">Waiting for Robinhood Chain…</div><a href="${EXPL}/tx/${hash}" target="_blank">tx ↗</a>`, 0);
    const rc = await p.waitForTransactionReceipt({ hash, timeout: 90_000 });
    if (rc.status !== 'success') throw new Error('Transaction reverted');
    toast(`<b><i></i>${esc(label)} — done</b><a href="${EXPL}/tx/${hash}" target="_blank">View tx ↗</a>`);
    return rc;
  } catch (e) {
    const msg = e.shortMessage || e.message || String(e);
    toast(`<b style="color:var(--down)">${esc(label)} failed</b><div class="m">${esc(/insufficient funds/i.test(msg) ? 'insufficient funds for gas — bridge a little ETH to Robinhood Chain.' : msg.slice(0, 160))}</div>`, 9000);
    throw e;
  }
}
const ERC20 = [
  { type: 'function', name: 'approve', stateMutability: 'nonpayable', inputs: [{ name: 's', type: 'address' }, { name: 'a', type: 'uint256' }], outputs: [{ type: 'bool' }] },
  { type: 'function', name: 'transfer', stateMutability: 'nonpayable', inputs: [{ name: 't', type: 'address' }, { name: 'a', type: 'uint256' }], outputs: [{ type: 'bool' }] },
  { type: 'function', name: 'allowance', stateMutability: 'view', inputs: [{ type: 'address' }, { type: 'address' }], outputs: [{ type: 'uint256' }] },
  { type: 'function', name: 'balanceOf', stateMutability: 'view', inputs: [{ type: 'address' }], outputs: [{ type: 'uint256' }] },
  { type: 'function', name: 'decimals', stateMutability: 'view', inputs: [], outputs: [{ type: 'uint8' }] },
];
async function ensureAllowance(token, spender, amount, label = 'Approve') {
  const { p, account } = await wallet();
  const cur = await p.readContract({ address: token, abi: ERC20, functionName: 'allowance', args: [account, spender] });
  if (cur >= amount) return;
  await tx(label, (w) => w.writeContract({ address: token, abi: ERC20, functionName: 'approve', args: [spender, amount] }));
}

// ------------------------------------------------------------------ SVG chart
// series: [{pts:[[t,v]...], color, width, fill}] ; marks: [{t, a, r, p}] ; base: number (dashed line)
function chart(el, series, { marks = [], base = null, fmt = fmtUsd, xfmt = fmtDT } = {}) {
  if (!el) return;
  const all = series.flatMap((s) => s.pts);
  if (all.length < 2) { el.innerHTML = '<div class="empty">Not enough data yet.</div>'; return; }
  const W = el.clientWidth || 600, H = el.clientHeight || 200, pl = 4, pr = 4, pt = 10, pb = 22;
  const t0 = Math.min(...all.map((p) => p[0])), t1 = Math.max(...all.map((p) => p[0]));
  const vals = all.map((p) => p[1]).concat(base != null ? [base] : []);
  let mn = Math.min(...vals), mx = Math.max(...vals);
  const padv = (mx - mn) * 0.08 || 1; mn -= padv; mx += padv;
  const X = (t) => pl + ((t - t0) / (t1 - t0 || 1)) * (W - pl - pr);
  const Y = (v) => pt + (1 - (v - mn) / (mx - mn || 1)) * (H - pt - pb);
  const path = (pts) => pts.map((p, i) => `${i ? 'L' : 'M'}${X(p[0]).toFixed(1)},${Y(p[1]).toFixed(1)}`).join('');
  const valAt = (pts, t) => { let best = pts[0]; for (const p of pts) { if (p[0] <= t) best = p; else break; } return best[1]; };
  let svg = `<svg viewBox="0 0 ${W} ${H}" preserveAspectRatio="none"><defs><linearGradient id="g${el.id}" x1="0" x2="0" y1="0" y2="1"><stop offset="0" stop-color="#ffb020" stop-opacity=".28"/><stop offset="1" stop-color="#ffb020" stop-opacity="0"/></linearGradient></defs>`;
  if (base != null) svg += `<line x1="${pl}" x2="${W - pr}" y1="${Y(base)}" y2="${Y(base)}" stroke="#5f5a50" stroke-dasharray="4 4"/>`;
  series.forEach((s, i) => {
    if (s.fill) svg += `<path d="${path(s.pts)}L${X(s.pts[s.pts.length - 1][0])},${H - pb}L${X(s.pts[0][0])},${H - pb}Z" fill="url(#g${el.id})"/>`;
    svg += `<path d="${path(s.pts)}" fill="none" stroke="${s.color}" stroke-width="${s.width || 1.8}" stroke-linejoin="round" stroke-linecap="round" ${i === 0 ? 'class="draw"' : ''}/>`;
  });
  const main = series[0].pts;
  for (const m of marks) {
    if (m.t < t0 || m.t > t1) continue;
    const x = X(m.t), y = Y(valAt(main, m.t));
    const col = m.a === 'buy' || m.a === 'cover' ? '#3ccf86' : m.a === 'short' ? '#ff6b5a' : '#f3efe6';
    const shape = m.a === 'buy' ? `<path d="M${x},${y - 7} l5,8 h-10z" fill="${col}"/>` : m.a === 'short' ? `<path d="M${x},${y + 7} l5,-8 h-10z" fill="${col}"/>` : `<circle cx="${x}" cy="${y}" r="3.6" fill="${col}"/>`;
    svg += `<g class="mk" data-tip="${esc(`${m.a.toUpperCase()} ${m.p ? fmtPx(m.p) : ''} · ${m.r || ''} · ${xfmt(m.t)}`)}" data-x="${x}" data-y="${y}">${shape}<circle cx="${x}" cy="${y}" r="9" fill="transparent"/></g>`;
  }
  const ticks = [t0, (t0 + t1) / 2, t1];
  svg += ticks.map((t, i) => `<text x="${X(t)}" y="${H - 5}" fill="#8f897c" font-size="10" font-family="JetBrains Mono" text-anchor="${i === 0 ? 'start' : i === 2 ? 'end' : 'middle'}">${xfmt(t)}</text>`).join('');
  svg += `<line class="hair" x1="0" x2="0" y1="${pt}" y2="${H - pb}" stroke="#3a362f" opacity="0"/></svg><div class="tip"></div>`;
  el.innerHTML = svg;
  const tip = $('.tip', el), hair = $('.hair', el), s = $('svg', el);
  $$('.mk', el).forEach((g) => {
    g.addEventListener('mouseenter', () => { tip.textContent = g.dataset.tip; tip.style.left = (g.dataset.x / W) * 100 + '%'; tip.style.top = (g.dataset.y / H) * 100 + '%'; tip.style.opacity = 1; });
    g.addEventListener('mouseleave', () => (tip.style.opacity = 0));
  });
  s.addEventListener('mousemove', (e) => {
    if (e.target.closest('.mk')) return;
    const r = s.getBoundingClientRect();
    const x = ((e.clientX - r.left) / r.width) * W;
    const t = t0 + ((x - pl) / (W - pl - pr)) * (t1 - t0);
    hair.setAttribute('x1', x); hair.setAttribute('x2', x); hair.setAttribute('opacity', 1);
    tip.innerHTML = series.map((sr) => `<span style="color:${sr.color}">${fmt(valAt(sr.pts, t))}</span>`).join(' · ') + ` <span class="muted">${xfmt(t)}</span>`;
    tip.style.left = (x / W) * 100 + '%'; tip.style.top = '12%'; tip.style.opacity = 1;
  });
  s.addEventListener('mouseleave', () => { hair.setAttribute('opacity', 0); tip.style.opacity = 0; });
  const d = $('.draw', el);
  if (d && d.getTotalLength) { const L = d.getTotalLength(); d.style.strokeDasharray = L; d.style.strokeDashoffset = L; d.getBoundingClientRect(); d.style.transition = 'stroke-dashoffset 1.2s ease'; d.style.strokeDashoffset = 0; }
}

function countUp(el, to) {
  const from = Number(el.dataset.v || 0); el.dataset.v = to;
  const t0 = performance.now();
  const tick = (now) => { const k = Math.min(1, (now - t0) / 900); el.textContent = Math.round(from + (to - from) * (1 - (1 - k) ** 3)).toLocaleString('en-US'); if (k < 1) requestAnimationFrame(tick); };
  requestAnimationFrame(tick);
}

  fS.innerHTML = Object.entries(strategies).map(([k, s]) => `<option value="${k}">${s.label}</option>`).join('');
  const marketOpts = (mode) => {
    const groups = mode === 'live' ? [['Live vault markets', catalog.live.map((id) => { const [k, s] = id.split(':'); const e = catalog[k].find((x) => x.sym === s); return [id, `${s} · ${e?.name || s}`]; })]]
      : mode.startsWith('hl') ? [['Perpetuals', catalog.perp.map((x) => [`perp:${x.sym}`, `${x.sym}-PERP · ${x.name}`])]]
      : [['Crypto', catalog.crypto.map((x) => [`crypto:${x.sym}`, `${x.sym}${x.name !== x.sym ? ' · ' + x.name : ''}`])], ['Perpetuals', catalog.perp.map((x) => [`perp:${x.sym}`, `${x.sym}-PERP${x.name !== x.sym ? ' · ' + x.name : ''}`])], ['Stocks & ETFs', catalog.stock.map((x) => [`stock:${x.sym}`, `${x.sym}${x.name !== x.sym ? ' · ' + x.name : ''}`])]];
    const cur = fM.value;
    fM.innerHTML = groups.map(([g, opts]) => `<optgroup label="${g}">${opts.map(([v, t]) => `<option value="${v}">${esc(t)}</option>`).join('')}</optgroup>`).join('');
    if ([...fM.options].some((o) => o.value === cur)) fM.value = cur;
  };
  marketOpts('paper');
  fM.value = 'crypto:BTC';

  const paramsUI = () => {
    const s = strategies[fS.value];
    $('#fBlurb').textContent = s.blurb;
    $('#fParams').innerHTML = Object.entries(s.params).map(([k, [d, lo, hi, label]]) => `<div><label class="f">${label}</label><input class="inp" type="number" data-p="${k}" value="${d}" min="${lo}" max="${hi}" step="${k === 'sigma' || k === 'depth' ? 0.1 : 1}"></div>`).join('');
    $$('#fParams input').forEach((i) => i.addEventListener('input', schedule));
  };
  const readParams = () => {
    const p = {};
    $$('#fParams input').forEach((i) => (p[i.dataset.p] = Number(i.value)));
    for (const k of ['StopLoss', 'TakeProfit', 'MaxLoss24h', 'PositionPct']) p[k[0].toLowerCase() + k.slice(1)] = Number($('#p' + k).value);
    p.side = $('#fSide').value; p.leverage = Number($('#fLev').value);
    return p;
  };
  const setParams = (p) => {
    for (const [k, v] of Object.entries(p)) {
      const i = $(`#fParams [data-p="${k}"]`); if (i) i.value = v;
      const j = $('#p' + k[0].toUpperCase() + k.slice(1)); if (j) j.value = v;
    }
    if (p.side) $('#fSide').value = p.side;
    if (p.leverage) $('#fLev').value = String(p.leverage);
  };
  const kindOf = () => fM.value.split(':')[0];
  const syncPerp = () => { $('#perpBox').classList.toggle('hide', kindOf() !== 'perp' || fMode.value === 'live'); };

  // regime hint
  const regime = async () => {
    const r = $('#regime');
    r.innerHTML = '<span class="skel">Reading the last 7 days…</span>';
    try {
      const g = await getJSON(`/api/regime?market=${encodeURIComponent(fM.value)}`);
      if (!g) { r.innerHTML = '<p>Not enough recent data.</p>'; return; }
      const fit = g.trend === 'up' ? ['Momentum', 'Buy the dip in an uptrend'] : g.trend === 'down' ? ['RSI reversal', 'Momentum'] : ['Mean reversion', 'RSI reversal'];
      const note = g.trend === 'up' ? 'Trend-following strategies have room to work. Dip buying needs the trend to hold.' : g.trend === 'down' ? 'Long-only agents mostly sit in cash in a downtrend. RSI reversal trades short bounces, so use a stop loss. Momentum waits for a breakout and sits out the drop.' : 'Range-bound prices suit strategies that buy stretches and sell the average.';
      r.innerHTML = `<span class="muted">Last 7 days</span> <b class="${g.trend === 'up' ? 'up' : g.trend === 'down' ? 'down' : ''}">${g.trend === 'up' ? 'Uptrend' : g.trend === 'down' ? 'Downtrend' : 'Sideways'}</b> <span class="mono muted">${fmtPct(g.changePct)} · ${g.vol} volatility</span><div class="chips">${fit.map((f) => `<span class="chip">${f}</span>`).join('')}</div><p>${note} A description of recent prices, not a forecast.</p>`;
    } catch (_) { r.innerHTML = '<p>Market description unavailable right now.</p>'; }
  };

  // local backtest (runs in the browser)
  const barsCache = {};
  const loadBars = async (market, hours) => {
    const key = market + ':' + hours;
    if (!barsCache[key]) barsCache[key] = getJSON(`/api/candles?market=${encodeURIComponent(market)}&hours=${hours}`).then((r) => r.bars);
    return barsCache[key];
  };
  let lastBt = null;
  const runBt = async () => {
    const win = Number($('#fWindow').value);
    if (win > 720 && !holder) { $('#btMetrics').innerHTML = '<div class="hint" style="grid-column:1/-1">90 and 180-day backtests are for $ORLA holders (1,000,000+).</div>'; $('#btChart').innerHTML = ''; return; }
    const strat = fS.value;
    const p = E.sanitize(strat, readParams(), kindOf());
    $('#btTitle').textContent = `Backtest · last ${win / 24} days`;
    $('#btChart').innerHTML = '<div class="empty"><span class="spin"></span></div>';
    try {
      const bars = await loadBars(fM.value, win + E.lookback(strat, p) + 4);
      const r = E.backtest(strat, p, bars, win);
      lastBt = r;
      if (!r) throw new Error('Not enough price history for this market.');
      chart($('#btChart'), [{ pts: r.curve, color: '#ffb020', width: 2, fill: true }, { pts: r.hold, color: '#8f897c', width: 1.4 }], { marks: r.marks, base: 10000 });
      const M = [['Return', fmtPct(r.returnPct), cls(r.returnPct)], ['Just holding', fmtPct(r.holdPct), cls(r.holdPct)], ['vs hold', fmtPct(r.vsHoldPct), cls(r.vsHoldPct)], ['Max drawdown', r.maxDrawdownPct.toFixed(1) + '%', ''], ['Win rate', r.winRatePct.toFixed(0) + '%', ''], ['Trades', r.trades, ''], ['Sharpe', r.sharpe.toFixed(2), ''], ['Avg hold', r.avgHoldHours >= 48 ? (r.avgHoldHours / 24).toFixed(1) + 'd' : r.avgHoldHours.toFixed(1) + 'h', ''], ['Profit factor', r.profitFactor.toFixed(2), ''], ['Best trade', fmtPct(r.bestTradePct), cls(r.bestTradePct)], ['Worst trade', fmtPct(r.worstTradePct), cls(r.worstTradePct)], ['Time in market', r.timeInMarketPct.toFixed(0) + '%', '']];
      if (kindOf() === 'perp') M.push(['Funding', fmtUsd(-r.fundingPaid), cls(-r.fundingPaid)]);
      $('#btMetrics').innerHTML = M.map(([l, v, c]) => `<div><div class="l">${l}</div><div class="v ${c}">${v}</div></div>`).join('');
    } catch (e) { $('#btChart').innerHTML = `<div class="empty">${esc(e.message)}</div>`; $('#btMetrics').innerHTML = ''; }
  };
  let tmr;
  function schedule() { clearTimeout(tmr); tmr = setTimeout(runBt, 350); }

  // optimizer & comparison (browser-side)
  $('#optBtn').onclick = async () => {
    const out = $('#optOut');
    out.innerHTML = '<div class="bt"><span class="spin"></span> Searching settings on the first two-thirds of the window…</div>';
    await sleep(30);
    const win = Math.min(Number($('#fWindow').value), holder ? 4320 : 720);
    const strat = fS.value, base = readParams();
    const bars = await loadBars(fM.value, win + 410);
    const fitH = Math.floor(win * 2 / 3), testH = win - fitH;
    const fitBars = bars.slice(0, bars.length - testH);
    let best = null;
    for (const g of E.grid(strat)) {
      const p = E.sanitize(strat, { ...base, ...g }, kindOf());
      const r = E.backtest(strat, p, fitBars, fitH);
      if (!r) continue;
      const score = r.returnPct - Math.abs(Math.min(0, r.maxDrawdownPct)) * 0.3 + (r.trades ? 0 : -5);
      if (!best || score > best.score) best = { score, g, fit: r };
    }
    if (!best) { out.innerHTML = '<div class="bt">Not enough data.</div>'; return; }
    const test = E.backtest(strat, E.sanitize(strat, { ...base, ...best.g }, kindOf()), bars, testH);
    out.innerHTML = `<div class="bt"><div class="bt-top"><span class="kicker">Best settings</span><button class="btn sm primary" id="useBest">Use these</button></div>
    <p class="mono" style="margin:8px 0;font-size:12.5px">${Object.entries(best.g).map(([k, v]) => `${k} ${v}`).join(' · ')}</p>
    <div class="metrics"><div><div class="l">Fit (first ${Math.round(fitH / 24)}d)</div><div class="v ${cls(best.fit.returnPct)}">${fmtPct(best.fit.returnPct)}</div></div><div><div class="l">Unseen (last ${Math.round(testH / 24)}d)</div><div class="v ${cls(test?.returnPct)}">${fmtPct(test?.returnPct)}</div></div><div><div class="l">Hold (unseen)</div><div class="v ${cls(test?.holdPct)}">${fmtPct(test?.holdPct)}</div></div></div>
    <p class="note">Settings were chosen on the first part only. The unseen result is the honest one: if it is much worse, the fit was luck.</p></div>`;
    $('#useBest').onclick = () => { setParams(best.g); runBt(); out.innerHTML = ''; };
  };
  $('#cmpBtn').onclick = async () => {
    const out = $('#optOut');
    out.innerHTML = '<div class="bt"><span class="spin"></span> Comparing every strategy…</div>';
    await sleep(30);
    const win = Math.min(Number($('#fWindow').value), holder ? 4320 : 720);
    const bars = await loadBars(fM.value, win + 410);
    const base = readParams();
    const rows = Object.keys(strategies).map((s) => {
      const def = Object.fromEntries(Object.entries(strategies[s].params).map(([k, v]) => [k, v[0]]));
      const r = E.backtest(s, E.sanitize(s, { ...base, ...def }, kindOf()), bars, win);
      return { s, r };
    }).filter((x) => x.r).sort((a, b) => b.r.returnPct - a.r.returnPct);
    out.innerHTML = `<div class="bt"><div class="bt-top"><span class="kicker">Strategies on ${esc(mkLabel(fM.value))} · ${win / 24}d</span></div>${rows.map((x, i) => `<div class="trow" style="grid-template-columns:24px 1fr 80px 80px 60px"><span class="muted">${i + 1}</span><span>${strategies[x.s].label}</span><span class="${cls(x.r.returnPct)}">${fmtPct(x.r.returnPct)}</span><span class="muted">${x.r.trades} trades</span><button class="btn sm" data-s="${x.s}">Use</button></div>`).join('')}</div>`;
    $$('#optOut [data-s]').forEach((b) => (b.onclick = () => { fS.value = b.dataset.s; paramsUI(); runBt(); out.innerHTML = ''; }));
  };

  // mode switching
  const modeUI = async () => {
    const m = fMode.value;
    marketOpts(m.startsWith('hl') ? 'hl' : m);
    if (m === 'live' && !catalog.live.includes(fM.value)) fM.value = 'crypto:ETH';
    if (m.startsWith('hl') && kindOf() !== 'perp') fM.value = 'perp:BTC';
    syncPerp();
    $('#launchNote').textContent = m === 'paper' ? 'Paper mode · $10,000 virtual cash · connecting a wallet only signs a message' : m === 'live' ? 'Live mode · trades the USDG in your vault · long-only, 1x' : 'Hyperliquid · trades perps in your own account through an approved agent wallet';
    $('#hlBox').innerHTML = '';
    if (m.startsWith('hl')) hlBox(m.split('-')[1]);
    regime(); schedule(); launchLabel();
  };
  const hlBox = async (network) => {
    const box = $('#hlBox');
    if (!session) { box.innerHTML = `<div class="callout" style="margin-top:10px"><span>Sign in, then approve an agent wallet on Hyperliquid ${network}.</span><a href="#" data-connect>Connect →</a></div>`; return; }
    box.innerHTML = '<div class="callout" style="margin-top:10px"><span class="spin"></span></div>';
    try {
      const st = await getJSON(`/api/me/hl?network=${network}`);
      if (st.status === 'connected') {
        box.innerHTML = `<div class="callout" style="margin-top:10px"><span>Hyperliquid ${network} connected · agent ${short(st.agentAddress)}${st.account ? ` · account ${fmtUsd(st.account.accountValue)}` : ''}</span><a href="#" id="hlForget">Disconnect</a></div>`;
        $('#hlForget').onclick = async (e) => { e.preventDefault(); await postJSON('/api/me/hl/forget', { network }); hlBox(network); };
        return;
      }
      box.innerHTML = `<div class="callout" style="margin-top:10px;flex-direction:column;align-items:flex-start"><span>Approve agent wallet <b class="mono">${short(st.agentAddress)}</b> on Hyperliquid ${network}. It can place and close orders; it can never withdraw.</span><button class="btn sm primary" id="hlApprove">Approve on Hyperliquid</button></div>`;
      $('#hlApprove').onclick = async () => {
        try {
          if (!provider) provider = await pickProvider();
          const chainHex = await provider.request({ method: 'eth_chainId' });
          const nonce = Date.now();
          const action = { type: 'approveAgent', signatureChainId: chainHex, hyperliquidChain: network === 'mainnet' ? 'Mainnet' : 'Testnet', agentAddress: st.agentAddress, agentName: 'ORLA', nonce };
          const typed = { domain: { name: 'HyperliquidSignTransaction', version: '1', chainId: parseInt(chainHex, 16), verifyingContract: '0x0000000000000000000000000000000000000000' },
            types: { EIP712Domain: [{ name: 'name', type: 'string' }, { name: 'version', type: 'string' }, { name: 'chainId', type: 'uint256' }, { name: 'verifyingContract', type: 'address' }], 'HyperliquidTransaction:ApproveAgent': [{ name: 'hyperliquidChain', type: 'string' }, { name: 'agentAddress', type: 'address' }, { name: 'agentName', type: 'string' }, { name: 'nonce', type: 'uint64' }] },
            primaryType: 'HyperliquidTransaction:ApproveAgent', message: { hyperliquidChain: action.hyperliquidChain, agentAddress: action.agentAddress, agentName: 'ORLA', nonce } };
          const sig = await provider.request({ method: 'eth_signTypedData_v4', params: [session.address, JSON.stringify(typed)] });
          const signature = { r: '0x' + sig.slice(2, 66), s: '0x' + sig.slice(66, 130), v: parseInt(sig.slice(130, 132), 16) };
          await postJSON('/api/me/hl', { network, action, nonce, signature });
          toast('<b><i></i>Hyperliquid connected</b>');
          hlBox(network);
        } catch (e) { toast(`<b style="color:var(--down)">Approval failed</b><div class="m">${esc(e.shortMessage || e.message)}</div>`); }
      };
    } catch (e) { box.innerHTML = `<div class="callout" style="margin-top:10px"><span>${esc(e.message)}</span></div>`; }
  };

  const launchLabel = () => { $('#launchBtn').textContent = session ? 'Backtest & launch' : 'Connect wallet to launch'; };
  $('#launchBtn').onclick = async () => {
    $('#launchErr').textContent = '';
    try {
      await ensureSession();
      launchLabel();
      const mode = fMode.value;
      const body = { name: $('#fName').value.trim(), strategy: fS.value, market: fM.value, mode: mode.startsWith('hl') ? 'hl' : mode, hlNetwork: mode.startsWith('hl') ? mode.split('-')[1] : undefined, params: readParams(), sourceId: qs.get('copy') || undefined };
      if (!body.name) { $('#launchErr').textContent = 'Give your agent a name.'; $('#fName').focus(); return; }
      $('#launchBtn').disabled = true; $('#launchBtn').innerHTML = '<span class="spin"></span> Backtesting & launching…';
      const { agent } = await postJSON('/api/me/agents', body);
      toast(`<b><i></i>${esc(agent.name)} is live</b><div class="m">${agent.mode === 'paper' ? 'Paper mode' : 'Live'} · first decision logged</div><a href="/agent/?id=${agent.id}">Open its page →</a>`);
      location.href = `/agent/?id=${agent.id}`;
    } catch (e) { if (e.message !== 'cancelled') $('#launchErr').textContent = e.message; }
    finally { $('#launchBtn').disabled = false; launchLabel(); }
  };

  $('#exampleBtn').onclick = () => {
    fMode.value = 'paper'; modeUI(); fS.value = 'momentum'; paramsUI(); fM.value = 'crypto:BTC';
    setParams({ positionPct: 50, stopLoss: 8, maxLoss24h: 5 }); if (!$('#fName').value) $('#fName').value = 'First Light';
    regime(); runBt(); $('#studio').scrollIntoView({ behavior: 'smooth', block: 'start' });
  };
  $('#riskToggle').onclick = () => { const b = $('#riskBox'); b.classList.toggle('hide'); $('#riskToggle').textContent = (b.classList.contains('hide') ? '▸' : '▾') + ' Risk controls · 0 = off'; };

  fS.onchange = () => { paramsUI(); schedule(); };
  fM.onchange = () => { syncPerp(); regime(); schedule(); };
  fMode.onchange = modeUI;
  $('#fWindow').onchange = schedule;
  ['#fSide', '#fLev', '#pStopLoss', '#pTakeProfit', '#pMaxLoss24h', '#pPositionPct'].forEach((s) => $(s).addEventListener('input', schedule));

  paramsUI();
  // deep links: ?market=… or ?copy=<agentId>
  if (qs.get('market') && [...fM.options].some((o) => o.value === qs.get('market'))) fM.value = qs.get('market');
  if (qs.get('copy')) {
    try {
      const { agent } = await getJSON(`/api/agents/${qs.get('copy')}`);
      fS.value = agent.strategy; paramsUI(); fM.value = agent.market; setParams(agent.params);
      $('#fName').value = (agent.name + ' copy').slice(0, 32);
      toast(`<b>Setup copied from ${esc(agent.name)}</b><div class="m">Backtest it, then launch your own.</div>`);
    } catch (_) {}
  }
  if (qs.get('mode') === 'live') { const keep = fM.value; fMode.value = 'live'; await modeUI(); if (catalog.live.includes(keep)) fM.value = keep; $('#fName').value = ($('#fName').value.replace(/ copy$/, '') + ' live').slice(0, 32); }
  syncPerp(); regime(); runBt();

  // signed-in parts: holder, my agents, vault
  const signedIn = async () => {
    launchLabel();
    if (!session) return;
    try { holder = (await getJSON('/api/me/holder')).holder; } catch (_) {}
    myAgents(); vaultCard();
    if (fMode.value.startsWith('hl')) hlBox(fMode.value.split('-')[1]);
  };
  document.addEventListener('orla:session', signedIn);
  await loadSession();
  signedIn();
};

  <div class="row2" style="margin-top:12px"><div><label class="f">Deposit USDG</label><div style="display:flex;gap:8px"><input class="inp" id="vDep" type="number" placeholder="25"><button class="btn" id="vDepBtn">Deposit</button></div></div>

  const VA = a.abi.AgentVault;
  $('#vDepBtn').onclick = async () => { const { v: vi } = await wallet(); await tx('Deposit USDG', (w) => w.writeContract({ address: usdg.address, abi: ERC20, functionName: 'transfer', args: [v.vault, vi.parseUnits($('#vDep').value || '0', 6)] })); vaultCard(); };

pages.pools = async () => {

  const sim = async () => {
    try {
      const { bars } = await getJSON('/api/candles?market=crypto:ETH&hours=48');
      const last = bars.slice(-41);
      ethPx = last[last.length - 1].c;
      const moves = last.slice(1).map((b, i) => Math.abs(Math.log(b.c / last[i].c)) * 100);
      const vol = moves.slice(-24).reduce((x, y) => x + y, 0) / 24;
      const fee = Math.min(1, 0.05 + vol * 0.25);
      const mx = Math.max(...moves) || 1;
      $('#simBars').innerHTML = moves.map((m) => `<i style="height:${Math.max(4, (m / mx) * 100)}%"></i>`).join('');
      $('#simPx').textContent = fmtPx(ethPx); $('#simVol').textContent = vol.toFixed(2) + '%';
      const live = stats?.fee?.total;
      const f = live ?? fee;
      $('#simFeeTop').textContent = f.toFixed(2) + '%';
      const calc = () => { const v = Number($('#simIn').value) || 0; $('#simOut').value = ((v * (1 - f / 100)) / ethPx).toFixed(5); };
      $('#simIn').oninput = calc; calc();
    } catch (_) {}
  };
  const loadStats2 = async () => {
    try {
      stats = await getJSON('/api/pools/stats');
      if (!stats.enabled) {
        $('#poolStatus').textContent = 'launching soon';

        .map(([l, v, sm]) => `<div class="card pstat reveal in"><div class="l">${l}</div><div class="v">${v}</div><small>${sm}</small></div>`).join('');
      $('#pSwaps').innerHTML = s.latest.length ? s.latest.map((x) => `<div class="swaprow"><span class="tag ${x.buy ? 'green' : 'red'}">${x.buy ? 'Buy WETH' : 'Sell WETH'}</span><span>${fmtUsd(x.usd)}</span><span class="muted">fee ${(x.fee / 1e4).toFixed(2)}%</span><a class="muted" href="${EXPL}/tx/${x.tx}" target="_blank">${fmtDT(x.at)} ↗</a></div>`).join('') : '<div class="empty">No swaps yet.</div>';
      $('#hookLink').href = `${EXPL}/address/${s.pool.hook}`;
      $('#pBlock').textContent = s.block ? `Read from Robinhood Chain up to block ${Number(s.block).toLocaleString()}${s.pool.since ? ` · pool live since ${new Date(s.pool.since).toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' })}` : ''}.` : '';
      if (s.fee) $('#lpFee').textContent = `Hook fee now ${s.fee.total.toFixed(2)}% · LP ${s.fee.lp.toFixed(2)}%`;
    } catch (_) {}
  };
  await loadStats2(); sim();

  // swap box: hooked pool (v4, via OrlaSwapRouter) or Uniswap V3 (best fee tier)
  let dir = 'buy', venue = 'hook', quote = null;
  const swapBox = () => {
    const pool = stats?.pool;
    const hasHook = !!(pool && pool.router);
    if (!hasHook) venue = 'v3';
    const [inS, outS] = dir === 'buy' ? ['USDG', 'WETH'] : ['WETH', 'USDG'];
    $('#swapBox').innerHTML = `<div style="display:flex;justify-content:space-between;align-items:center"><h3>Swap · WETH / USDG</h3><button class="flip" id="flip">${inS} → ${outS} ⇄</button></div>
    ${hasHook ? `<div class="chips" style="margin-top:10px"><button class="chip ${venue === 'hook' ? 'on' : ''}" data-v="hook">Hooked pool</button><button class="chip ${venue === 'v3' ? 'on' : ''}" data-v="v3">Uniswap V3</button></div>` : ''}
    <label class="f">You pay (${inS})</label><input class="inp mono" id="swIn" placeholder="0.00" inputmode="decimal">
    <div class="box" style="margin-top:10px;border:1px solid var(--line);border-radius:10px;padding:10px 12px"><small class="muted">You receive (≈)</small><div class="mono" id="swOut" style="font-size:18px">–</div></div>

    $('#flip').onclick = () => { dir = dir === 'buy' ? 'sell' : 'buy'; swapBox(); };
    $$('#swapBox [data-v]').forEach((b) => (b.onclick = () => { venue = b.dataset.v; swapBox(); }));
    let qt;
    $('#swIn').oninput = () => { clearTimeout(qt); qt = setTimeout(doQuote, 300); };
    $('#swGo').onclick = doSwap;
  };
  const QV3 = [{ type: 'function', name: 'quoteExactInputSingle', stateMutability: 'nonpayable', inputs: [{ type: 'tuple', name: 'p', components: [{ name: 'tokenIn', type: 'address' }, { name: 'tokenOut', type: 'address' }, { name: 'amountIn', type: 'uint256' }, { name: 'fee', type: 'uint24' }, { name: 'sqrtPriceLimitX96', type: 'uint160' }] }], outputs: [{ name: 'amountOut', type: 'uint256' }, { type: 'uint160' }, { type: 'uint32' }, { type: 'uint256' }] }];
  const QV4 = [{ type: 'function', name: 'quoteExactInputSingle', stateMutability: 'nonpayable', inputs: [{ type: 'tuple', name: 'p', components: [{ name: 'poolKey', type: 'tuple', components: [{ name: 'currency0', type: 'address' }, { name: 'currency1', type: 'address' }, { name: 'fee', type: 'uint24' }, { name: 'tickSpacing', type: 'int24' }, { name: 'hooks', type: 'address' }] }, { name: 'zeroForOne', type: 'bool' }, { name: 'exactAmount', type: 'uint128' }, { name: 'hookData', type: 'bytes' }] }], outputs: [{ name: 'amountOut', type: 'uint256' }, { name: 'gasEstimate', type: 'uint256' }] }];

    const amt = $('#swIn').value;

    $('#swOut').innerHTML = '<span class="spin"></span>';

      const amountIn = v.parseUnits(amt, dIn);
      if (venue === 'hook') {
        const key = { currency0: WETH.address, currency1: USDG.address, fee: 0x800000, tickSpacing: 60, hooks: stats.pool.hook };
        const { result } = await p.simulateContract({ address: a.uni.v4Quoter, abi: QV4, functionName: 'quoteExactInputSingle', args: [{ poolKey: key, zeroForOne: dir === 'sell', exactAmount: amountIn, hookData: '0x' }] });

      } else {
        let best = null;
        for (const fee of [100, 500, 3000, 10000]) {
          try { const { result } = await p.simulateContract({ address: a.uni.v3Quoter, abi: QV3, functionName: 'quoteExactInputSingle', args: [{ tokenIn: tIn.address, tokenOut: tOut.address, amountIn, fee, sqrtPriceLimitX96: 0n }] }); if (!best || result[0] > best.out) best = { fee, out: result[0] }; } catch (_) {}
        }
        if (!best) throw new Error('No liquidity for this amount.');
        quote = { amountIn, out: best.out, fee: best.fee, tIn, tOut, dOut };
      }
      $('#swOut').textContent = `${fmtN(Number(v.formatUnits(quote.out, quote.dOut)), 6)} ${dir === 'buy' ? 'WETH' : 'USDG'}${quote.fee ? ` · ${quote.fee / 1e4}% tier` : ''}`;

    if (!lp.enabled) { box.innerHTML = '<h3>WETH / USDG LP vault</h3><p class="hint">The hooked pool is not deployed yet.</p>'; return; }
    const pool = lp.pool;

New vaults get the bot as operator.</p>${c.agentVaultFactory ? `<pre>${c.agentVaultFactory}</pre>` : btns('avf', 'Deploy')}</div>
    <div class="card step"><h3>2 · PayrollFactory ${done('payrollFactory')}</h3><p class="hint">USDG payroll vaults.</p>${c.payrollFactory ? `<pre>${c.payrollFactory}</pre>` : btns('pf', 'Deploy')}</div>
    <div class="card step"><h3>3 · OrlaHook ${done('hook')}</h3><p class="hint">Uniswap v4 hook (fee 0.05% → 1%, +0.005% per tick). Its address must encode its permissions, so a salt is mined in your browser and it is deployed through the standard CREATE2 deployer.</p>${c.hook ? `<pre>${c.hook}</pre>` : btns('hook', 'Mine salt & deploy')}</div>
    <div class="card step"><h3>4 · OrlaLPVaultFactory ${done('lpFactory')}</h3>${c.lpFactory ? `<pre>${c.lpFactory}</pre>` : btns('lpf', 'Deploy')}</div>
    <div class="card step"><h3>5 · OrlaSwapRouter ${done('swapRouter')}</h3>${c.swapRouter ? `<pre>${c.swapRouter}</pre>` : btns('router', 'Deploy')}</div>
    <div class="card step"><h3>6 · Register & initialize the WETH/USDG pool ${done('poolInitialized')}</h3><p class="hint">Registers the pool on the hook (you become its owner) and initializes it at the current ETH price. Liquidity comes from LP vaults on the Pools page.</p>${c.poolInitialized ? '<pre>initialized</pre>' : (c.hook ? btns('pool', 'Register & initialize') : '<button class="btn primary" disabled>Register & initialize</button>')}</div>
    <div class="card step"><h3>7 · Token & links</h3><label class="f">$ORLA contract address</label><input class="inp mono" id="sTok" value="${esc(c.token || '')}" placeholder="0x… (after launch)"><div class="row2"><div><label class="f">X URL</label><input class="inp" id="sX" value="${esc(c.x || '')}" placeholder="https://x.com/…"></div><div><label class="f">GitHub URL</label><input class="inp" id="sGh" value="${esc(c.github || '')}" placeholder="https://github.com/…"></div></div><button class="btn" id="sSave" style="margin-top:12px">Save</button></div>
    ${claimCard}
    <div class="card step"><h3>Engine</h3><p class="hint">The engine runs every 5 minutes by itself (Cron Trigger). You can also run it now.</p><button class="btn" id="sRun">Run jobs now</button><pre id="sOut" class="hide"></pre></div>`;
    const save = (o) => postJSON('/api/setup/save', o);
    const deploy = async (label, abiX, bytecode, args) => {
      const { w, p } = await wallet();
      toast(`<b><span class="spin"></span> ${label}</b><div class="m">Confirm in your wallet…</div>`, 0);
      // wallets sometimes cap contract creations at ~1.2M gas: set the limit ourselves
      const v = await V();
      const data = v.encodeDeployData({ abi: abiX, bytecode, args });
      const est = await p.estimateGas({ account: w.account.address, data }).catch(() => null);
      const floor = BigInt(Math.ceil(((data.length - 2) / 2) * 260 + 1_500_000));
      const gas = est ? (est * 13n) / 10n > floor ? (est * 13n) / 10n : floor : floor;
      const hash = await w.sendTransaction({ data, gas });
      const rc = await p.waitForTransactionReceipt({ hash, timeout: 120_000 });
      if (rc.status !== 'success' || !rc.contractAddress) throw new Error('Deployment failed');
      toast(`<b><i></i>${label} deployed</b><div class="m">${rc.contractAddress}</div>`);
      return { address: rc.contractAddress, block: Number(rc.blockNumber) };
    };
    const T = info.tokens;
    $$('[data-d]').forEach((b) => (b.onclick = async () => {
      try {
        b.disabled = true;
        const k = b.dataset.d;
        if (!info.bot && ['avf', 'lpf'].includes(k)) throw new Error('Set BOT_PRIVATE_KEY first: the factory needs the bot address.');
        if (k === 'avf') {
          const r = await deploy('AgentVaultFactory', a.abi.AgentVaultFactory, B.AgentVaultFactory, [a.uni.swapRouter02, info.bot, T.map((t) => t.address), T.map((t) => t.feed), T.map((t) => t.maxStaleness), T.map((t) => t.feeTier || 0)]);
          await save({ agentVaultFactory: r.address });
        }
        if (k === 'pf') { const r = await deploy('PayrollFactory', a.abi.PayrollFactory, B.PayrollFactory, [T.find((t) => t.symbol === 'USDG').address]); await save({ payrollFactory: r.address }); }
        if (k === 'lpf') { const r = await deploy('OrlaLPVaultFactory', a.abi.OrlaLPVaultFactory, B.OrlaLPVaultFactory, [a.uni.poolManager, info.bot]); await save({ lpFactory: r.address }); }
        if (k === 'router') { const r = await deploy('OrlaSwapRouter', a.abi.OrlaSwapRouter, B.OrlaSwapRouter, [a.uni.poolManager]); await save({ swapRouter: r.address }); }
        if (k === 'hook') {
          const v = await V();
          const init = v.concat([B.OrlaHook, v.encodeAbiParameters([{ type: 'address' }, { type: 'uint24' }, { type: 'uint24' }, { type: 'uint24' }], [a.uni.poolManager, 500, 10000, 50])]);
          const initHash = v.keccak256(init);
          const FLAGS = 0x20c4, MASK = 0x3fff;
          toast('<b><span class="spin"></span> Mining a hook salt…</b><div class="m">Usually a few seconds.</div>', 0);
          let salt, addr;
          const dep = a.uni.create2.toLowerCase().slice(2);
          for (let i = 0; i < 2_000_000; i++) {
            salt = '0x' + i.toString(16).padStart(64, '0');
            const h = v.keccak256('0xff' + dep + salt.slice(2) + initHash.slice(2));
            if ((parseInt(h.slice(-4), 16) & MASK) === FLAGS) { addr = v.getAddress('0x' + h.slice(-40)); break; }
            if (i % 5000 === 0) await sleep(0);
          }
          if (!addr) throw new Error('No salt found');
          const { w, p } = await wallet();
          toast(`<b><span class="spin"></span> Deploying OrlaHook</b><div class="m">${addr}</div>`, 0);
          const hdata = v.concat([salt, init]);
          const hest = await p.estimateGas({ account: w.account.address, to: a.uni.create2, data: hdata }).catch(() => null);
          const hfloor = BigInt(Math.ceil(((hdata.length - 2) / 2) * 260 + 1_500_000));
          const hgas = hest ? (hest * 13n) / 10n > hfloor ? (hest * 13n) / 10n : hfloor : hfloor;
          const hash = await w.sendTransaction({ to: a.uni.create2, data: hdata, gas: hgas });
          await p.waitForTransactionReceipt({ hash, timeout: 120_000 });
          const code = await p.getCode({ address: addr });
          if (!code || code === '0x') throw new Error('Hook not found at ' + addr);
          await save({ hook: addr, hookSalt: salt });
          toast(`<b><i></i>OrlaHook deployed</b><div class="m">${addr}</div>`);
        }
        if (k === 'pool') {
          const v = await V();
          const key = { currency0: T.find((t) => t.symbol === 'WETH').address, currency1: T.find((t) => t.symbol === 'USDG').address, fee: 0x800000, tickSpacing: 60, hooks: c.hook };
          const { p } = await wallet();
          const pc = await p.readContract({ address: c.hook, abi: a.abi.OrlaHook, functionName: 'pools', args: [v.keccak256(v.encodeAbiParameters([{ type: 'address' }, { type: 'address' }, { type: 'uint24' }, { type: 'int24' }, { type: 'address' }], [key.currency0, key.currency1, key.fee, key.tickSpacing, key.hooks]))] });
          if (/^0x0+$/.test(pc[0])) await tx('Register pool', (w) => w.writeContract({ address: c.hook, abi: a.abi.OrlaHook, functionName: 'registerPool', args: [key, '0x0000000000000000000000000000000000000000', 0] }));
          const { bars } = await getJSON('/api/candles?market=crypto:ETH&hours=24');
          const eth = bars[bars.length - 1].c;
          const sqrt = BigInt(Math.floor(Math.sqrt(eth * 1e-12) * 2 ** 48)) * 2n ** 48n; // sqrt(price) * 2^96, USDG(6) per WETH(18)
          const PM = [{ type: 'function', name: 'initialize', stateMutability: 'nonpayable', inputs: [{ name: 'key', type: 'tuple', components: [{ name: 'currency0', type: 'address' }, { name: 'currency1', type: 'address' }, { name: 'fee', type: 'uint24' }, { name: 'tickSpacing', type: 'int24' }, { name: 'hooks', type: 'address' }] }, { name: 'sqrtPriceX96', type: 'uint160' }], outputs: [{ type: 'int24' }] }];
          const rc = await tx('Initialize pool', (w) => w.writeContract({ address: a.uni.poolManager, abi: PM, functionName: 'initialize', args: [key, sqrt] }));
          await save({ poolInitialized: true, poolBlock: Number(rc.blockNumber), poolSince: Date.now() });
        }
        draw();
      } catch (e) { toast(`<b style="color:var(--down)">Setup step failed</b><div class="m">${esc(e.shortMessage || e.message)}</div>`, 10000); b.disabled = false; }
    }));
    $$('[data-b]').forEach((b) => (b.onclick = async () => {
      const k = b.dataset.b;
      try {
        $$('[data-b],[data-d]').forEach((x) => (x.disabled = true));
        let salt;
        if (k === 'hook') {
          const v = await V();
          const init = v.concat([B.OrlaHook, v.encodeAbiParameters([{ type: 'address' }, { type: 'uint24' }, { type: 'uint24' }, { type: 'uint24' }], [a.uni.poolManager, 500, 10000, 50])]);
          const initHash = v.keccak256(init), dep = a.uni.create2.toLowerCase().slice(2);
          toast('<b><span class="spin"></span> Mining a hook salt…</b><div class="m">Usually a few seconds.</div>', 0);
          for (let i = 0; i < 2_000_000; i++) {
            const sx = '0x' + i.toString(16).padStart(64, '0');
            const h = v.keccak256('0xff' + dep + sx.slice(2) + initHash.slice(2));
            if ((parseInt(h.slice(-4), 16) & 0x3fff) === 0x20c4) { salt = sx; break; }
            if (i % 5000 === 0) await sleep(0);
          }
          if (!salt) throw new Error('No salt found');
        }
        toast('<b><span class="spin"></span> The bot is deploying…</b><div class="m">No wallet popup. This can take up to a minute.</div>', 0);
        const r = await postJSON('/api/setup/bot', { step: k, salt });
        toast(`<b><i></i>Done</b><div class="m">${esc(r.address || 'pool ready')}${r.pendingAdmin ? ' · accept admin in step 8' : ''}</div>`, 8000);
        draw();
      } catch (e) { toast(`<b style="color:var(--down)">Setup step failed</b><div class="m">${esc(e.message)}</div>`, 12000); $$('[data-b],[data-d]').forEach((x) => (x.disabled = false)); }
    }));
    $$('[data-claim]').forEach((b) => (b.onclick = async () => {
      try { b.disabled = true; await tx('Accept admin', (w) => w.writeContract({ address: b.dataset.claim, abi: ADM, functionName: 'acceptAdmin', args: [], gas: 120000n })); draw(); }
      catch (_) { b.disabled = false; }
    }));
    $('#sSave').onclick = async () => { await save({ token: $('#sTok').value.trim() || null, x: $('#sX').value.trim() || null, github: $('#sGh').value.trim() || null }); toast('<b><i></i>Saved</b>'); };
    $('#sRun').onclick = async () => { $('#sOut').classList.remove('hide'); $('#sOut').textContent = 'running…'; try { const r = await postJSON('/api/setup/run'); $('#sOut').textContent = JSON.stringify(r.health, null, 2); } catch (e) { $('#sOut').textContent = e.message; } };
  };
  document.addEventListener('orla:session', draw);
  await loadSession();
  draw();
};

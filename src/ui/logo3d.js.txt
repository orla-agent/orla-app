// ORLA home: the mark as a floating 3D object (three.js), extruded from the real logo paths.
// Falls back to the flat SVG when WebGL is unavailable or reduced motion is requested.
const stage = document.getElementById('logoStage');
const canvas = document.getElementById('logo3d');
const reduced = matchMedia('(prefers-reduced-motion: reduce)').matches;
const hasGL = (() => { try { const c = document.createElement('canvas'); return !!(c.getContext('webgl2') || c.getContext('webgl')); } catch (_) { return false; } })();
if (stage && canvas && !reduced && hasGL) init().catch((e) => console.warn('3D logo off:', e));

async function init() {
  const THREE = await import('https://esm.sh/three@0.160.0');
  const { SVGLoader } = await import('https://esm.sh/three@0.160.0/examples/jsm/loaders/SVGLoader.js');
  const { RoomEnvironment } = await import('https://esm.sh/three@0.160.0/examples/jsm/environments/RoomEnvironment.js');
  const svg = document.querySelector('#logoStage svg.fallback');
  const paths = [...svg.querySelectorAll('path')].map((p) => ({ d: p.getAttribute('d'), dot: p.getAttribute('fill') !== 'currentColor' }));

  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  renderer.outputColorSpace = THREE.SRGBColorSpace;
  renderer.toneMapping = THREE.ACESFilmicToneMapping;
  renderer.toneMappingExposure = 1.05;
  const scene = new THREE.Scene();
  const pmrem = new THREE.PMREMGenerator(renderer);
  scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
  const camera = new THREE.PerspectiveCamera(32, 1, 0.1, 100);
  camera.position.set(0, 0, 11);

  const body = new THREE.MeshPhysicalMaterial({ color: 0xf1ece2, roughness: 0.32, metalness: 0.05, clearcoat: 0.6, clearcoatRoughness: 0.25 });
  const amber = new THREE.MeshPhysicalMaterial({ color: 0xffb020, roughness: 0.25, metalness: 0.1, clearcoat: 0.8, emissive: 0x6a3d00, emissiveIntensity: 0.25 });
  const group = new THREE.Group();
  const loader = new SVGLoader();
  let dotMesh = null;
  for (const p of paths) {
    const data = loader.parse(`<svg xmlns="http://www.w3.org/2000/svg"><path d="${p.d}"/></svg>`);
    for (const sp of data.paths) {
      for (const shape of SVGLoader.createShapes(sp)) {
        const geo = new THREE.ExtrudeGeometry(shape, { depth: p.dot ? 70 : 90, bevelEnabled: true, bevelThickness: 10, bevelSize: 8, bevelSegments: 6, curveSegments: 10 });
        const m = new THREE.Mesh(geo, p.dot ? amber : body);
        if (p.dot) { dotMesh = m; m.position.z = 10; }
        group.add(m);
      }
    }
  }
  // center & scale; SVG y points down
  const box = new THREE.Box3().setFromObject(group);
  const c = box.getCenter(new THREE.Vector3());
  group.children.forEach((m) => m.geometry.translate(-c.x, -c.y, -45));
  const holder = new THREE.Group();
  group.scale.set(0.0062, -0.0062, 0.0062);
  holder.add(group);
  scene.add(holder);

  scene.add(new THREE.AmbientLight(0xffffff, 0.25));
  const key = new THREE.DirectionalLight(0xffffff, 2.2); key.position.set(4, 5, 6); scene.add(key);
  const rim = new THREE.PointLight(0xffa020, 40, 30); rim.position.set(-4, -2, 3); scene.add(rim);
  const rim2 = new THREE.PointLight(0xffd27a, 18, 30); rim2.position.set(5, -3, -2); scene.add(rim2);

  const resize = () => { const r = stage.getBoundingClientRect(); renderer.setSize(r.width, r.height, false); camera.aspect = r.width / r.height; camera.updateProjectionMatrix(); };
  resize(); addEventListener('resize', resize);
  let mx = 0, my = 0, tx = 0, ty = 0;
  addEventListener('pointermove', (e) => { mx = (e.clientX / innerWidth - 0.5) * 2; my = (e.clientY / innerHeight - 0.5) * 2; });
  let visible = true;
  new IntersectionObserver(([e]) => (visible = e.isIntersecting)).observe(stage);
  stage.classList.add('webgl');
  const t0 = performance.now();
  const loop = (now) => {
    requestAnimationFrame(loop);
    if (!visible) return;
    const t = (now - t0) / 1000;
    tx += (mx - tx) * 0.05; ty += (my - ty) * 0.05;
    holder.rotation.y = Math.sin(t * 0.45) * 0.45 + tx * 0.35;
    holder.rotation.x = Math.sin(t * 0.32) * 0.12 + ty * 0.2 - 0.05;
    holder.rotation.z = Math.sin(t * 0.25) * 0.04;
    holder.position.y = Math.sin(t * 0.8) * 0.12;
    if (dotMesh) { dotMesh.position.z = 10 + Math.sin(t * 1.6) * 18; }
    rim.position.x = Math.cos(t * 0.5) * 5; rim.position.y = Math.sin(t * 0.5) * 3;
    renderer.render(scene, camera);
  };
  requestAnimationFrame(loop);
}
